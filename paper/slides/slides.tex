\begin{frame}{The problem: the list and the pack never meet}
\small
\begin{itemize}
\item CDSCO publishes monthly lists of batches that failed laboratory tests (NSQ) or were declared spurious.
\item A patient holds a strip with a batch number on the foil. Checking means finding the month and comparing by hand.
\item QR codes (top 300 brands) prove where a pack came from. They carry no test result.
\end{itemize}
\vfill
\begin{center}

\bignum{\ShelfTwelvePlusPct\%}{of flagged batches still have $\geq$12 months of shelf life at the alert}
\end{center}
\vfill
\end{frame}

\begin{frame}{The decision: is this exact pack from a listed batch?}
\begin{center}
\footnotesize
\renewcommand{\arraystretch}{1.3}\begin{tabularx}{0.92\textwidth}{@{}l X@{}}
\toprule
\textcolor{brick}{\V{FLAGGED}} & Confirmed batch matches an NSQ entry, and product plus dates or maker agree \\
\textcolor{brick}{\V{SUSPECTED\_FAKE}} & Confirmed batch matches an entry listed as purported spurious \\
\V{NOT\_LISTED} & Nothing survives exact and misread-aware search of the lists loaded. Never a certificate, never green \\
\textcolor{muted}{\V{CANT\_VERIFY}} & Evidence too weak after bounded steps and two questions: the agent abstains \\
\bottomrule
\end{tabularx}
\end{center}
\vspace{1mm}
\footnotesize
\begin{itemize}
\item Advisory only: every result says ``show this pack to your pharmacist''. Never ``safe'', never ``stop taking''.
\item Every alarm cites the list, month, failed test and source row. Only the batch is flagged, never the brand.
\end{itemize}
\end{frame}

% 4
\begin{frame}{Why a lookup script is not enough}
\small
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{\color{teal}Reading fails confidently}
\begin{itemize}
\item OCR read a scuffed \texttt{B} as \texttt{3} at confidence 1.0; the enhanced re-read agreed.
\item An exact lookup then misses the listed batch.
\end{itemize}
\vspace{2mm}
\textbf{\color{teal}Batch numbers collide}
\begin{itemize}
\item \CollisionKeys{} keys are shared by unrelated products.
\item \ShortKeyPct\% of keys have under four characters.
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{\color{brick}Siblings turn misreads into alarms}
\begin{itemize}
\item \SiblingKeys{} listed keys have a same-maker neighbour one confusable character away (\texttt{R21119} / \texttt{R21719}).
\item A misread can land exactly on the sibling, and every other field agrees.
\end{itemize}
\vspace{2mm}
\textbf{\color{teal}Fields do not line up}
\begin{itemize}
\item Strips name the marketer, lists the manufacturer.
\item Spurious entries list the maker as ``under investigation''.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Data: CDSCO's portal, ingested with provenance}
\small
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{itemize}
\item Portal JSON back to \FirstMonth: \NumNsqRows{} NSQ rows over \NumMonths{} months and \NumSpuriousTabRows{} spurious rows (snapshot of \SnapshotDate).
\item Every response stored with its SHA-256; every alert cites its source row.
\item Imports are atomic and idempotent; bad rows are quarantined with reasons, never typed in by hand.
\end{itemize}
\end{column}
\begin{column}{0.47\textwidth}
\small
\textbf{Quirks handled}
\begin{itemize}
\item Latin-1 encoding, dashes arriving as ``?''
\item brand present in only \BrandPresentPct\% of rows
\item multi-batch cells, GS1 prefixes, dates in batch cells
\item maker and marketer in one cell
\item \MissingDatePct\% of rows without a manufacturing date
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\scriptsize

\begin{itemize}
\item Packs rendered from real alert rows: clean, QR, glare, misread, spurious, collision, unlisted.

\item Eleven authored diagnostic scenarios run offline in the test suite.
\end{itemize}
\textbf{Arms}
\begin{itemize}
\item A: lookup script (QR/OCR, regex, exact hit = alarm)
\item B: agent, rule planner; C: agent, hybrid planner
\item each with an oracle user and with every question skipped
\end{itemize}
\end{column}
\begin{column}{0.46\textwidth}
\textbf{Metrics}
\begin{itemize}
\item hit recall, type accuracy
\item false-alarm rate, overall and on collisions
\item abstention, questions per pack
\item median and p95 latency
\item safety violations (must be 0)
\end{itemize}
Wilson 95\% intervals; exact McNemar for paired arms.
\end{column}
\end{columns}

\begin{frame}{A/B decision, surprises and limitations}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\footnotesize
\textbf{Preregistered rule R-A}\\
Adopt the agent if collision false alarms drop by $\geq$50\% (relative) and hit recall falls by $\leq$5 points.\\[2mm]
\OutcomeText
\end{column}
\begin{column}{0.46\textwidth}
\small
\textbf{Limitations}
\begin{itemize}
\item \V{NOT\_LISTED} only covers sampled, loaded lists
\item all packs are synthetic; no real strip photos yet; English labels only
\item the oracle answers perfectly; real people may not
\item small packs per scenario; laptop CPU latency
\end{itemize}
\end{column}
\end{columns}
\end{frame}
